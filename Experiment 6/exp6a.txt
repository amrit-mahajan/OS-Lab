# ==============================================================================
# Experiment 6(a): User Session Duration & Login Tracker Shell Script
# Algorithm: Queries 'who' and extracts login timestamp using 'awk'
# ==============================================================================

#!/bin/bash

read -p "Enter username: " user
login_time=$(who | awk -v u="$user" '$1 == u {print $3, $4; exit}')

if [ -z "$login_time" ]; then
    echo "user is not currently logged in."
else
    echo "Login time: $login_time"
fi
